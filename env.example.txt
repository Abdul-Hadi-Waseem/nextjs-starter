# Backend base URL. Example: https://api.yoursite.com
# Locally this points at the app's own mock API (src/app/api).
# REQUIRED (dev and build).

# The mock API answers 404 in production. Set to true only for a demo
# deployment: its login accepts ANY password and "admin…" emails become admin.

# true (default) = /dashboard requires the session cookie; without it you're
# sent to /auth/sign-in. Set false only for an open demo.
NEXT_PUBLIC_AUTH_ENABLED=true

# Name of the httpOnly cookie your backend sets after login.
AUTH_COOKIE_NAME=access_token

# Who resizes next/image images:

#               every pod runs sharp and caches separately: CPU/memory spikes.
#   cloudflare  Cloudflare Image Transformations (/cdn-cgi/image/...) in production.
#   none        no resizing; sharp never runs. Images are served as committed.
NEXT_PUBLIC_IMAGE_LOADER=
# Optional origin for Cloudflare transforms, e.g. https://yoursite.com (default: same origin).
NEXT_PUBLIC_CF_IMAGE_BASE=

# Public URL of this app, for absolute Open Graph image URLs (optional).
